cmake_minimum_required(VERSION 3.24...4.4)
project(WinampOgg VERSION 1.2.1 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC with -A Win32 for 32-bit Winamp.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(CMAKE_POLICY_DEFAULT_CMP0069 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
include(CheckIPOSupported)
check_ipo_supported(RESULT ogg_ipo OUTPUT ogg_ipo_error LANGUAGES C CXX)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE "${ogg_ipo}")
add_compile_definitions("$<$<CONFIG:Debug>:WINAMP_OPUS_DEBUG_CRT>")
add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:/FI${CMAKE_CURRENT_SOURCE_DIR}/cmake/dynamic_crt.h>")
option(OGG_INPUT_BUILD_TESTS "Build independent playback tests" ON)
option(OGG_INPUT_ENABLE_SPEEX "Enable Ogg Speex decoding" ON)
option(OGG_INPUT_BUILD_OPUS "Build sibling in_opus plugin as well" ON)
set(OGG_INPUT_OPUS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../winamp" CACHE PATH "Path to GPL in_opus source")
include(cmake/Dependencies.cmake)
option(OGG_INPUT_DEPENDENCIES_ONLY "Build audio libraries while developing the plugin" OFF)
if(OGG_INPUT_DEPENDENCIES_ONLY)
    add_library(ogg_core_check OBJECT src/stream.cpp src/credentials.cpp src/probe.cpp src/metadata.cpp src/audio.cpp src/opus_reader.cpp src/plugin.cpp src/extended_read.cpp)
    target_include_directories(ogg_core_check PRIVATE src)
    target_compile_definitions(ogg_core_check PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX FLAC__NO_DLL)
    target_compile_options(ogg_core_check PRIVATE /W4 /permissive- /utf-8)
    target_link_libraries(ogg_core_check PRIVATE opusfile speex_resampler ogg_vorbisfile FLAC ogg_speex)
    target_sources(ogg_core_check PRIVATE src/vorbis_reader.cpp src/flac_reader.cpp src/ui.cpp src/settings.cpp src/tag_writer.cpp src/dialogs.rc)
    add_library(ogg_test_compile OBJECT tests/fixtures.cpp tests/codec_fixtures.cpp tests/ogg_tests.cpp tests/plugin_tests.cpp tests/network_tests.cpp tests/dialog_tests.cpp)
    target_include_directories(ogg_test_compile PRIVATE src tests)
    target_compile_definitions(ogg_test_compile PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX FLAC__NO_DLL)
    target_compile_options(ogg_test_compile PRIVATE /W4 /permissive- /utf-8)
    target_link_libraries(ogg_test_compile PRIVATE opusfile speex_resampler ogg_vorbisenc FLAC ogg_speex)
    add_executable(ogg_stream_tests tests/network_tests.cpp tests/fixtures.cpp tests/codec_fixtures.cpp src/stream.cpp src/credentials.cpp src/probe.cpp)
    target_include_directories(ogg_stream_tests PRIVATE src tests)
    target_compile_definitions(ogg_stream_tests PRIVATE OGG_STREAM_ONLY UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX FLAC__NO_DLL)
    target_compile_options(ogg_stream_tests PRIVATE /W4 /permissive- /utf-8)
    target_link_libraries(ogg_stream_tests PRIVATE opusfile speex_resampler ogg_vorbisenc FLAC ogg_speex winhttp credui ole32 ws2_32)
    return()
endif()
add_library(ogg_audio STATIC src/stream.cpp src/credentials.cpp src/probe.cpp src/metadata.cpp src/audio.cpp
    src/opus_reader.cpp src/opus_native.cpp src/ambisonics.cpp src/vorbis_reader.cpp src/flac_reader.cpp
    src/channel_layout.cpp src/tag_writer.cpp src/processing.cpp)
if(OGG_INPUT_ENABLE_SPEEX)
    target_sources(ogg_audio PRIVATE src/speex_reader.cpp)
    target_compile_definitions(ogg_audio PUBLIC OGG_HAVE_SPEEX)
endif()
target_include_directories(ogg_audio PUBLIC src)
target_compile_definitions(ogg_audio PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX FLAC__NO_DLL
    OGG_OPUS_HAVE_OSCE=1 OGG_OPUS_HAVE_OSCE_BWE=1)
target_compile_options(ogg_audio PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(ogg_audio PUBLIC opusfile ogg_vorbisfile FLAC ogg_speex speex_resampler winhttp credui ole32)
add_library(in_vorbis SHARED src/plugin.cpp src/extended_read.cpp src/ui.cpp src/settings.cpp
    src/matroska.cpp src/host_settings.cpp src/language.cpp src/dialogs.rc src/version.rc src/in_vorbis.def)
target_link_libraries(in_vorbis PRIVATE ogg_audio user32 gdi32 advapi32 comctl32)
target_compile_options(in_vorbis PRIVATE /W4 /permissive- /utf-8)
set_target_properties(in_vorbis PROPERTIES PREFIX "" OUTPUT_NAME "in_vorbis"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
install(TARGETS in_vorbis RUNTIME DESTINATION Plugins COMPONENT Runtime)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md FEATURE_PARITY.md VERIFICATION.md DESTINATION ogg-input COMPONENT Runtime)
install(FILES docs/LANGUAGE_PACKS.md docs/ADVANCED_DECODING.md DESTINATION ogg-input/docs COMPONENT Runtime)
install(FILES localization/en-US.rc DESTINATION ogg-input/localization COMPONENT Runtime)
if(PROJECT_IS_TOP_LEVEL)
    install(FILES docs/PACKAGE_README.md DESTINATION . RENAME README.md COMPONENT Runtime)
    install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION . COMPONENT Runtime)
endif()
install(DIRECTORY "${CMAKE_BINARY_DIR}/ogg-licenses/" DESTINATION licenses COMPONENT Runtime)
if(OGG_INPUT_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
